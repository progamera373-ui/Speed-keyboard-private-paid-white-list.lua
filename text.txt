

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local Workspace = game:GetService("Workspace")
local UIS = game:GetService("UserInputService")
local Player = Players.LocalPlayer

local cacheLavas = {}
local cachePillieres = {}
local cacheMovingWalls = {}
local cacheVentiladores = {}
local ultimaActualizacionCache = 0
local INTERVALO_CACHE = 5


local BypassActive = {
    ScreenGui = nil,
    HeartbeatConnection = nil,
    Active = true
}

function BypassActive:ProtegerGUI(gui)
    self.ScreenGui = gui
    -- Evitar que el GUI sea destruido
    gui.Destroying:Connect(function()
        if self.Active then
            local nuevo = gui:Clone()
            nuevo.Parent = Player.PlayerGui
            self.ScreenGui = nuevo
        end
    end)
    
    gui.Changed:Connect(function(prop)
        if prop == "Parent" and gui.Parent ~= Player.PlayerGui and self.Active then
            gui.Parent = Player.PlayerGui
        end
    end)
end

function BypassActive:IniciarAntiDetection()
    self.HeartbeatConnection = RunService.Heartbeat:Connect(function()
        if not self.ScreenGui or not self.ScreenGui.Parent then
            self:ProtegerGUI(self.ScreenGui or BypassActive.ScreenGui)
        end
        
        if #cacheLavas == 0 and #cachePillieres == 0 and #cacheMovingWalls == 0 then
            RebuildCache()
        end
    end)
end

function BypassActive:Desactivar()
    self.Active = false
    if self.HeartbeatConnection then
        self.HeartbeatConnection:Disconnect()
    end
end


local screenGui = Instance.new("ScreenGui")
screenGui.Name = "CreditoGui"
screenGui.ResetOnSpawn = false
screenGui.Parent = Player.PlayerGui

local frame = Instance.new("Frame")
frame.Size = UDim2.new(0, 160, 0, 40)
frame.Position = UDim2.new(0.5, -80, 1, -50)
frame.BackgroundColor3 = Color3.fromRGB(15, 15, 15)
frame.BackgroundTransparency = 0.3
frame.BorderSizePixel = 0
frame.Parent = screenGui

local corner = Instance.new("UICorner")
corner.CornerRadius = UDim.new(0, 10)
corner.Parent = frame

local stroke = Instance.new("UIStroke")
stroke.Color = Color3.fromRGB(255, 215, 0)
stroke.Thickness = 2

label.Size = UDim2.new(1, 0, 1, 0)
label.Position = UDim2.new(0, 0, 0, 0)
label.BackgroundTransparency = 1
label.Text = "NEXA HUB"
label.TextColor3 = Color3.fromRGB(255, 215, 0)

label.Font = Enum.Font.GothamBold
label.TextSize = 14
label.Parent = frame

BypassActive:ProtegerGUI(screenGui)
BypassActive:IniciarAntiDetection()


local function ClasificarObjeto(obj)
    if not (obj:IsA("BasePart") or obj:IsA("MeshPart")) then return end
    local nombre = obj.Name
    local nombreL = nombre:lower()
    
    
    if nombreL:find("lava") or nombreL:find("kill") or nombreL:find("danger") 
       or nombreL:find("daño") or nombreL:find("dead") then
        table.insert(cacheLavas, obj)
    end
    
    
    if nombreL:find("pillier") or nombreL:find("barre") or nombre == "WallAndFloor" then
        table.insert(cachePillieres, obj)
    end
    
    
    local numMatch = nombre:match("^MovingWall(%d+)$")
    if numMatch then
        local num = tonumber(numMatch)
        if num and num >= 1 and num <= 12 then
            table.insert(cacheMovingWalls, obj)
        end
    end
    
   
    if nombre == "Circle" or nombre == "Circle.001" or nombre == "Circle.002" then
        table.insert(cacheVentiladores, obj)
    end
end

local function RebuildCache()
    -- Limpiar caches
    cacheLavas = {}
    cachePillieres = {}
    cacheMovingWalls = {}
    cacheVentiladores = {}

    for _, obj in pairs(Workspace:GetDescendants()) do
        ClasificarObjeto(obj)
    end
    ultimaActualizacionCache = tick()
end

local function ParalizarWallAndFloor()
    for _, obj in pairs(cachePillieres) do
        if obj and obj.Parent and obj.Name == "WallAndFloor" then
            pcall(function()
                obj.Anchored = true
                obj.CanCollide = true
                obj.Transparency = 0
                obj.CanTouch = false
                -- Desactivar scripts hijos
                for _, s in pairs(obj:GetChildren()) do
                    if s:IsA("Script") or s:IsA("LocalScript") or s:IsA("ModuleScript") then
                        s.Disabled = true
                    end
                end
            end)
        end
    end
end

local function DestruirVentiladores()
    for _, obj in pairs(cacheVentiladores) do
        if obj and obj.Parent then
            pcall(function()
                local modelo = obj.Parent
                if modelo and modelo:IsA("Model") then
                    modelo:Destroy()
                else
                    obj:Destroy()
                end
            end)
        end
    end
    cacheVentiladores = {}
end

local function EliminarNPCs()
    pcall(function()
        for _, obj in pairs(Workspace:GetDescendants()) do
            if obj:IsA("Humanoid") and obj.Parent ~= Player.Character then
                local modelo = obj.Parent
                if modelo and modelo.Parent then
                    modelo:Destroy()
                end
            end
        end
    end)
end

local function EliminarMovingWalls()
    for i = #cacheMovingWalls, 1, -1 do
        local obj = cacheMovingWalls[i]
        if obj and obj.Parent then
            pcall(function() obj:Destroy() end)
        end
        table.remove(cacheMovingWalls, i)
    end
end

local function LimpiarLavas()
    local c = Player.Character
    local root = c and c:FindFirstChild("HumanoidRootPart")
    if not root then return end
    local alturaRef = root.Position.Y
    
    for i = #cacheLavas, 1, -1 do
        local obj = cacheLavas[i]
        if obj and obj.Parent and obj.Position.Y < (alturaRef + 12) then
            pcall(function() obj:Destroy() end)
            table.remove(cacheLavas, i)
        end
    end
end

local function EliminarPiques()
    -- Solo buscar Piques/TriggerPart sin afectar otros objetos
    pcall(function()
        for _, obj in pairs(Workspace:GetDescendants()) do
            if (obj.Name == "Piques" or obj.Name == "TriggerPart") and obj.Parent then
                obj:Destroy()
            end
        end
    end)
end

UIS.JumpRequest:Connect(function()
    local char = Player.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum and hum.Health > 0 then
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
        end
    end
end)


local ultimoHeartbeat = 0
RunService.Heartbeat:Connect(function()
    local ahora = tick()
    if ahora - ultimoHeartbeat < 0.2 then return end -- Throttle a 5 veces/seg
    ultimoHeartbeat = ahora
    
    for _, obj in pairs(cachePillieres) do
        if obj and obj.Parent then
            if obj.Transparency > 0 or not obj.CanCollide or not obj.Anchored then
                pcall(function()
                    obj.Anchored = true
                    obj.CanCollide = true
                    obj.Transparency = 0
                    obj.CanTouch = false
                end)
            end
        end
    end
end)


task.spawn(function()
    while true do
        task.wait(2.5)
        local ahora = tick()
        
        
        if ahora - ultimaActualizacionCache >= INTERVALO_CACHE then
            RebuildCache()
        end
        
        
        LimpiarLavas()
        EliminarMovingWalls()
        EliminarNPCs()
        EliminarPiques()
        DestruirVentiladores()
    end
end)


Player.CharacterAdded:Connect(function()
    task.wait(0.5)
    RebuildCache()
    ParalizarWallAndFloor()
    LimpiarLavas()
    EliminarMovingWalls()
    EliminarNPCs()
    EliminarPiques()
    DestruirVentiladores()
end)


RebuildCache()
ParalizarWallAndFloor()
LimpiarLavas()
EliminarMovingWalls()
EliminarNPCs()
EliminarPiques()
DestruirVentiladores()
